export function cleanSteamText(text: string): string {
  const entities: Record<string, string> = {
    quot: '"',
    amp: "&",
    apos: "'",
    lt: "<",
    gt: ">",
    nbsp: " ",
    ndash: "–",
    mdash: "—",
    hellip: "…",
    rsquo: "’",
    lsquo: "‘",
    ldquo: "“",
    rdquo: "”",
  };
  return text
    .replace(/&(#x[\da-f]+|#\d+|[a-z]+);/gi, (entity, code: string) => {
      if (!code.startsWith("#")) return entities[code.toLowerCase()] ?? entity;
      const point = code.toLowerCase().startsWith("#x")
        ? Number.parseInt(code.slice(2), 16)
        : Number.parseInt(code.slice(1), 10);
      return point > 0 && point <= 0x10ffff ? String.fromCodePoint(point) : "";
    })
    .replace(/\[img\][\s\S]*?\[\/img\]/gi, "")
    .replace(/\[\/?[a-z*][^\]]*\]/gi, "")
    .replace(/<[^>]*>/g, " ")
    .replace(/(?:https?:\/\/|www\.)[^\s<>\]]+/gi, "")
    .replace(/\s+/g, " ")
    .trim();
}

export function classifyAndCleanSteamNews(item: {
  title: string;
  contents: string;
  tags?: string[];
  feedlabel?: string;
}): { category: "patch_note" | "event" | "general"; cleanSummary: string } {
  const cleanSummary = cleanSteamText(item.contents).slice(0, 280);

  const text = [
    item.title,
    cleanSummary,
    ...(item.tags ?? []),
    item.feedlabel ?? "",
  ].join(" ");
  const isPatch =
    /\b(patch\w*|hotfix\w*|update\w*|changelog\w*|balance\w*|bug\w*|fix\w*|version\w*|notes|v\d+)\b/i.test(
      text,
    );

  if (isPatch) return { category: "patch_note", cleanSummary };

  const isEvent =
    /\b(event\w*|double\s*xp|free\s*weekend|season\w*|tournament\w*|drops|raid\w*|festival\w*|limited[\s-]*time|anniversary|dlc|expansion\w*)\b/i.test(
      text,
    );

  if (isEvent) return { category: "event", cleanSummary };

  return { category: "general", cleanSummary };
}

export function evaluateDealAlert(params: {
  isOwned: boolean;
  currentPrice: number;
  targetPrice: number | null;
  historicalLow: number;
}): { shouldAlert: boolean; reason: "TARGET_MET" | "ALL_TIME_LOW" | null } {
  if (params.isOwned) return { shouldAlert: false, reason: null };
  if (
    params.targetPrice !== null &&
    params.currentPrice <= params.targetPrice
  ) {
    return { shouldAlert: true, reason: "TARGET_MET" };
  }
  if (params.currentPrice > 0 && params.currentPrice <= params.historicalLow) {
    return { shouldAlert: true, reason: "ALL_TIME_LOW" };
  }
  return { shouldAlert: false, reason: null };
}

import type { TRPCRouterRecord } from "@trpc/server";
import { z } from "zod/v4";

import { and, asc, eq, sql } from "@acme/db";
import { userBacklog } from "@acme/db/schema";
import {
  BacklogExportSchema,
  DealItemSchema,
  SearchDealsInputSchema,
  SteamFeedInputSchema,
  UpsertBacklogItemSchema,
} from "@acme/validators";

import {
  fetchDeals,
  fetchGameMetadata,
  fetchGamePrices,
  fetchSteamFeed,
} from "../steam-discovery";
import { protectedProcedure, publicProcedure } from "../trpc";

export const backlogRouter = {
  getSteamFeed: publicProcedure
    .input(SteamFeedInputSchema)
    .query(async ({ input }) => {
      const feed = await fetchSteamFeed(input.steamAppId ?? 1091500);
      return {
        patchNotes:
          input.category === "event"
            ? []
            : feed.patchNotes.slice(0, input.limit),
        events:
          input.category === "patch_note"
            ? []
            : feed.events.slice(0, input.limit),
      };
    }),

  searchDeals: publicProcedure
    .input(SearchDealsInputSchema)
    .output(z.array(DealItemSchema))
    .query(({ input }) => fetchDeals(input?.query, input)),

  getGamePrices: publicProcedure
    .input(
      z.object({ steamAppId: z.number().int().positive().max(2147483647) }),
    )
    .query(async ({ input }) => ({
      prices: await fetchGamePrices(input.steamAppId),
    })),

  removeBacklogItem: protectedProcedure
    .input(
      z.object({ steamAppId: z.number().int().positive().max(2147483647) }),
    )
    .mutation(async ({ ctx, input }) => {
      await ctx.db
        .delete(userBacklog)
        .where(
          and(
            eq(userBacklog.userId, ctx.session.user.id),
            eq(userBacklog.steamAppId, input.steamAppId),
          ),
        );
      return { removed: input.steamAppId };
    }),

  getGameDetails: publicProcedure
    .input(
      z.object({ steamAppId: z.number().int().positive().max(2147483647) }),
    )
    .query(async ({ input }) => {
      const metadata = fetchGameMetadata(input.steamAppId);
      const [feed, game, prices] = await Promise.all([
        fetchSteamFeed(input.steamAppId),
        metadata,
        fetchGamePrices(input.steamAppId, metadata),
      ]);
      return {
        steamAppId: input.steamAppId,
        feed,
        metadata: game,
        prices,
      };
    }),

  getBacklog: protectedProcedure.query(async ({ ctx }) => {
    const items = await ctx.db
      .select()
      .from(userBacklog)
      .where(eq(userBacklog.userId, ctx.session.user.id))
      .orderBy(asc(userBacklog.title));
    return items.map((item) => ({
      ...item,
      amountPaid: item.amountPaid === null ? null : Number(item.amountPaid),
      targetPrice: item.targetPrice === null ? null : Number(item.targetPrice),
    }));
  }),

  importBacklogJson: protectedProcedure
    .input(BacklogExportSchema)
    .mutation(async ({ ctx, input }) => {
      await ctx.db.transaction(async (tx) => {
        for (const item of input.items) {
          const values = {
            ...item,
            amountPaid: item.amountPaid?.toFixed(2) ?? null,
            targetPrice: item.targetPrice?.toFixed(2) ?? null,
            completionStatus: item.completionStatus ?? "backlog",
            notifyPatches: item.notifyPatches ?? true,
            notifyEvents: item.notifyEvents ?? true,
          };
          await tx
            .insert(userBacklog)
            .values({ ...values, userId: ctx.session.user.id })
            .onConflictDoUpdate({
              target: [userBacklog.userId, userBacklog.steamAppId],
              set: values,
            });
        }
      });
      return { imported: input.items.length };
    }),

  getBacklogStats: protectedProcedure.query(async ({ ctx }) => {
    const [stats] = await ctx.db
      .select({
        totalSpent: sql<string>`coalesce(sum(${userBacklog.amountPaid}) filter (where ${userBacklog.isOwned}), 0)`,
        ownedCount:
          sql<number>`count(*) filter (where ${userBacklog.isOwned})`.mapWith(
            Number,
          ),
        wishlistedCount:
          sql<number>`count(*) filter (where not ${userBacklog.isOwned})`.mapWith(
            Number,
          ),
      })
      .from(userBacklog)
      .where(eq(userBacklog.userId, ctx.session.user.id));
    return {
      totalSpent: Number(stats?.totalSpent ?? 0),
      ownedCount: stats?.ownedCount ?? 0,
      wishlistedCount: stats?.wishlistedCount ?? 0,
    };
  }),

  upsertBacklogItem: protectedProcedure
    .input(UpsertBacklogItemSchema)
    .mutation(async ({ ctx, input }) => {
      // Undefined fields stay unchanged on conflict; null explicitly clears a price.
      const values = {
        ...input,
        amountPaid:
          input.amountPaid == null
            ? input.amountPaid
            : input.amountPaid.toFixed(2),
        targetPrice:
          input.targetPrice == null
            ? input.targetPrice
            : input.targetPrice.toFixed(2),
      };
      const [item] = await ctx.db
        .insert(userBacklog)
        .values({ ...values, userId: ctx.session.user.id })
        .onConflictDoUpdate({
          target: [userBacklog.userId, userBacklog.steamAppId],
          set: values,
        })
        .returning();
      return item;
    }),

  exportBacklogJson: protectedProcedure
    .output(BacklogExportSchema)
    .query(async ({ ctx }) => {
      const items = await ctx.db
        .select({
          steamAppId: userBacklog.steamAppId,
          title: userBacklog.title,
          isOwned: userBacklog.isOwned,
          amountPaid: userBacklog.amountPaid,
          targetPrice: userBacklog.targetPrice,
          completionStatus: userBacklog.completionStatus,
          notifyPatches: userBacklog.notifyPatches,
          notifyEvents: userBacklog.notifyEvents,
        })
        .from(userBacklog)
        .where(eq(userBacklog.userId, ctx.session.user.id))
        .orderBy(asc(userBacklog.steamAppId));
      return {
        version: 1 as const,
        exportedAt: new Date().toISOString(),
        items: items.map((item) => ({
          ...item,
          amountPaid:
            item.amountPaid === null ? undefined : Number(item.amountPaid),
          targetPrice:
            item.targetPrice === null ? undefined : Number(item.targetPrice),
        })),
      };
    }),
} satisfies TRPCRouterRecord;

function readFilters(): Filters {

        ...defaults,
        ...input.data,
        minDiscount: [0, 25, 50, 75].includes(Number(params.get("discount")))
          ? Number(params.get("discount"))
          : 0,
        hideOwned: params.get("owned") === "1",

}
function writeFilters(value: Filters) {
  const params = new URLSearchParams();
  if (value.query) params.set("q", value.query);
  if (value.storeIds.length)
    params.set("stores", [...value.storeIds].sort().join(","));
  if (value.maxPrice !== undefined) params.set("price", String(value.maxPrice));
  if (value.steamOnly) params.set("steam", "1");
  if (value.onSale) params.set("sale", "1");
  if (value.hideOwned) params.set("owned", "1");
  if (value.minDiscount) params.set("discount", String(value.minDiscount));
  if (value.sort !== "deals") params.set("sort", value.sort);
  if (value.page) params.set("page", String(value.page));
  if (value.pageSize !== 24) params.set("size", String(value.pageSize));
  if (value.view === "list") params.set("view", "list");
  window.history.replaceState(
    null,
    "",
    `${window.location.pathname}${params.size ? `?${params}` : ""}`,
  );
}

type Prices = RouterOutputs["backlog"]["getGameDetails"]["prices"];
function alertFor(item: BacklogItem, prices: Prices | undefined) {
  return prices?.currentPrice == null
    ? { shouldAlert: false, reason: null }
    : evaluateDealAlert({
        isOwned: item.isOwned,
        currentPrice: prices.currentPrice,
        targetPrice: item.targetPrice ?? null,
        historicalLow: prices.historicalLow ?? -1,
      });
}

function WatchCard({
  item,
  prices,
  pending,
}: {
  item: BacklogItem;
  prices: Prices | undefined;
  pending: boolean;
}) {

  const [message, setMessage] = useState("");
  const [saving, setSaving] = useState(false);
  const alert = alertFor(item, prices);
  async function update(change: Partial<BacklogItem>) {
    setSaving(true);
    try {
      await save({
        steamAppId: item.steamAppId,
        title: item.title,
        isOwned: false,
        ...change,
      });
      setMessage("Price watch updated.");
    } catch {
      setMessage("Unable to save changes. Please try again.");
    } finally {
      setSaving(false);
    }
  }
  return (
    <article className="game-card">
      <GameArt appId={item.steamAppId} title={item.title} />
      <div className="space-y-5 p-5">
        <h2 className="text-xl font-semibold">{item.title}</h2>
        <span className="watch-state" data-alert-state={alert.reason ?? "WAIT"}>
          {alert.reason === "ALL_TIME_LOW"
            ? "All-time low"
            : alert.reason === "TARGET_MET"
              ? "Target met"
              : "Watching for a drop"}
        </span>
        <dl className="grid grid-cols-2 gap-4 text-sm">
          <div>
            <dt className="text-muted-foreground">Current Best Price ($)</dt>
